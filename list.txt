function hasTwoCubeSums(n) {
	//TODO
  let count = 0;
  let max = Math.ceil(Math.cbrt(n));
  for(let i = 0; i < max; i += 1)
    {
      let x = Math.cbrt(n - (i*i*i));
      if(x%1 == 0) count +=1
    }
  return count >= 4
}

function ipv4Parser(ip, mask){
  //your code here
  let IPsp = ip.split(".");
  let MASKsp = mask.split(".");
  
  let ipP = [];
  let maskP = [];
  for(let i = 0;i < 4;i++)
    {
      ipP[i] = Number(IPsp[i]) & Number(MASKsp[i]);
      maskP[i] = Number(IPsp[i]) & ~Number(MASKsp[i]);
    }
  return [ipP.join('.'), maskP.join('.')];
}

function whatCentury(year)
{
   let century = Number(year);
   century = Math.ceil(year * 0.01);
   if(century < 20)
     {
       return century + "th";
     }
  switch(century % 10)
    {
      case 1:
        return century + "st";
        break;
      
      case 2:
        return century + "nd";
        break;
        
      case 3:
        return century + "rd";
        break;
      
      default:
        return century + "th";
        break;
    }
}

function findMissing(list) {
  let missing = list[1] - list[0];
  for(let i = 1; i < list.length - 1; i++)
    {
      if(list[i+1] - list[i] < missing)
        {
          missing = list[i+1] - list[i];
        }
    }
  for(let i = 0; i < list.length - 1 ; i++)
    {
      if(list[i] + missing != list[i+1])
        {
          return list[i] + missing;
        }
    }
}

function primeFactors(n){
    let string = "";
  
  for(let i = 2; n != 1; i++)
    {
      if(n%i != 0)
        {
          continue;
        }
      let counter = 0;
      while(n % i === 0) {
        n = n / i;
        counter++;
      }
      if(counter > 1)
        {
          string = string + `(${i}**${counter})`;
        }
      else
        {
          string = string + `(${i})`;
        }
    }
  return string;
}
